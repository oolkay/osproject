# Program to calculate 5! (factorial of 5)
Begin Data Section
0 0    # Program counter
1 1000 # Stack pointer
2 0    # System call result
3 0    # Instruction count

# User variables
50 5   # n = 5 (the number to calculate factorial for)
51 1   # result = 1 (factorial result accumulator)
52 0   # temp - used for multiplication loop counter
53 0   # temp_result - used for multiplication
99 -1  # Always true condition for jumps
End Data Section

Begin Instruction Section
# Main program
0 SET 5 50    # Set n = 5
1 SET 1 51    # Set result = 1

# Main factorial loop - while n > 0
2 JIF 50 123   # If n <= 0, jump to end

# Multiply result by n (we multiply by adding n times)
3 SET 0 53    # temp_result = 0
4 CPY 50 52   # Set counter = n

# Multiplication loop - add result to temp_result counter times
5 JIF 52 9    # If counter <= 0, exit multiplication loop
6 ADDI 53 51  # temp_result += result
7 ADD 52 -1   # counter--
8 JIF 99 5    # Jump back to multiplication loop (using always true condition)

# Update result, decrement n and print
9 CPY 53 51   # result = temp_result
10 SYSCALL PRN 51 # Print current factorial
11 ADD 50 -1   # n--
12 SYSCALL PRN 50 # n
13 JIF 99 2    # Jump back to main loop (using always true condition)

# End program
123 SYSCALL PRN 51  # Print final result
124 HLT             # Halt execution
End Instruction Section
